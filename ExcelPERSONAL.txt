Option Explicit

' ===== 開いているプレゼンのコピーを Downloads に出力（元の拡張子を維持） =====
Public Sub ExportPresToDownloads()

    Dim pres As Presentation
    Dim folderPath As String, savePath As String, ext As String
    Dim fmt As PpSaveAsFileType
    Dim i As Long

    On Error GoTo ErrHandler

    If Application.Presentations.Count = 0 Then
        MsgBox "プレゼンテーションが開かれていません。", vbExclamation
        Exit Sub
    End If
    Set pres = ActivePresentation

    folderPath = GetDownloadsFolder()
    If Dir(folderPath, vbDirectory) = "" Then
        MsgBox "ダウンロードフォルダが見つかりません:" & vbCrLf & folderPath, vbExclamation
        Exit Sub
    End If

    ' 元の拡張子から保存形式を決める（未保存の新規ファイルは .pptx）
    i = InStrRev(pres.Name, ".")
    If i > 1 Then ext = LCase$(Mid$(pres.Name, i)) Else ext = ".pptx"

    Select Case ext
        Case ".pptm": fmt = ppSaveAsOpenXMLPresentationMacroEnabled
        Case ".ppsx": fmt = ppSaveAsOpenXMLShow
        Case ".ppt":  fmt = ppSaveAsPresentation
        Case Else:    ext = ".pptx": fmt = ppSaveAsOpenXMLPresentation
    End Select

    savePath = folderPath & "\" _
             & SanitizeFileName(BaseName(pres.Name)) & "_" _
             & Format(Now, "yyyymmdd_hhnnss") & ext

    ' 元ファイルのパスを一切参照しないため、OneDrive上でも動く
    pres.SaveCopyAs FileName:=savePath, FileFormat:=fmt

    Exit Sub

ErrHandler:
    MsgBox "出力に失敗しました。" & vbCrLf & Err.Number & ": " & Err.Description, vbExclamation
End Sub


' ===== 開いているプレゼンを PDF にして Downloads に出力 =====
Public Sub ExportPdfToDownloads()

    Dim pres As Presentation
    Dim folderPath As String, savePath As String

    On Error GoTo ErrHandler

    If Application.Presentations.Count = 0 Then
        MsgBox "プレゼンテーションが開かれていません。", vbExclamation
        Exit Sub
    End If
    Set pres = ActivePresentation

    folderPath = GetDownloadsFolder()
    If Dir(folderPath, vbDirectory) = "" Then
        MsgBox "ダウンロードフォルダが見つかりません:" & vbCrLf & folderPath, vbExclamation
        Exit Sub
    End If

    savePath = folderPath & "\" _
             & SanitizeFileName(BaseName(pres.Name)) & "_" _
             & Format(Now, "yyyymmdd_hhnnss") & ".pdf"

    pres.SaveCopyAs FileName:=savePath, FileFormat:=ppSaveAsPDF

    Exit Sub

ErrHandler:
    MsgBox "出力に失敗しました。" & vbCrLf & Err.Number & ": " & Err.Description, vbExclamation
End Sub


' ===== ダウンロードフォルダのパス取得 =====
Private Function GetDownloadsFolder() As String
    Dim p As String
    On Error Resume Next
    p = CreateObject("WScript.Shell").RegRead( _
        "HKCU\Software\Microsoft\Windows\CurrentVersion\Explorer\Shell Folders\" & _
        "{374DE290-123F-4565-9164-39C4925E467B}")
    On Error GoTo 0

    If Len(p) = 0 Then p = Environ$("USERPROFILE") & "\Downloads"
    If Right$(p, 1) = "\" Then p = Left$(p, Len(p) - 1)
    GetDownloadsFolder = p
End Function


' ===== 拡張子を除いたファイル名 =====
Private Function BaseName(ByVal s As String) As String
    Dim i As Long
    i = InStrRev(s, ".")
    If i > 1 Then s = Left$(s, i - 1)
    BaseName = s
End Function


' ===== ファイル名に使えない文字を置換 =====
Private Function SanitizeFileName(ByVal s As String) As String
    Dim bad As Variant, i As Long
    bad = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For i = LBound(bad) To UBound(bad)
        s = Replace(s, bad(i), "_")
    Next i
    SanitizeFileName = Trim$(s)
End Function
